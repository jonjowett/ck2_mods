expd_purchase_claims_notify_host_that_from_purchased_a_claim_on_this_title = {
	if = {
		# Only notify host if they either aren't the decision-taker, or if they're in SP with debug mode on
		limit = {
			OR = {
				AND = {
					multiplayer = yes
					ROOT_FROM = { is_multiplayer_host_character = no }
				}
				AND = {
					multiplayer = no
					has_global_flag = expd_purchase_claims_debug
				}
			}
		}
		custom_tooltip = {
			text = expd_purchase_claims_notify_host_that_from_purchased_a_claim_on_this_title_tt
			save_event_target_as = expd_purchase_claims_notification_title
			ROOT_FROM = { save_event_target_as = expd_purchase_claims_notification_purchaser }
			holder_scope = { save_event_target_as = expd_purchase_claims_notification_holder }
			any_player = {
				limit = {
					OR = {
						AND = {
							multiplayer = yes
							is_multiplayer_host_character = yes
						}
						AND = {
							multiplayer = no
							ai = no
							has_global_flag = expd_purchase_claims_debug
						}
					}
				}
				character_event = { id = expd_purchase_claims.1000 }
			}
		}
	}
}

expd_purchase_claims_notify_relevant_players_that_from_purchased_a_claim_on_this_title = {
	# If any of the following are players (who are not the purchaser), they are notified:
	#  - Current holder
	#  - Landed ruler in current holder's top liege's realm
	#  - Any other claimant
	# This obviously only matters in multiplayer.
	#  - In SP, the only player is the purchaser, so they wouldn't be notified.
	if = {
		limit = { multiplayer = yes }
		hidden_tooltip = {
			save_event_target_as = expd_purchase_claims_notification_title
			ROOT_FROM = { save_event_target_as = expd_purchase_claims_notification_purchaser }
			holder_scope = { save_event_target_as = expd_purchase_claims_notification_holder }
		}
		holder_scope = {
			if = {
				limit = { expd_purchase_claims_this_character_is_a_notifiable_player = yes }
				custom_tooltip = {
					text = expd_purchase_claims_this_player_is_notified_tt
					character_event = { id = expd_purchase_claims.1000 }
				}
			}
			top_liege = {
				if = {
					limit = {
						expd_purchase_claims_this_character_is_a_notifiable_player = yes
						NOT = { character = PREV }         # Not the title holder
					}
					custom_tooltip = {
						text = expd_purchase_claims_this_player_is_notified_tt
						character_event = { id = expd_purchase_claims.1000 }
					}
				}
				any_realm_character = {
					limit = {
						expd_purchase_claims_this_character_is_a_notifiable_player = yes
						NOT = { character = PREV }         # Not the top liege
						NOT = { character = PREVPREV }     # Not the title holder
					}
					custom_tooltip = {
						text = expd_purchase_claims_this_player_is_notified_tt
						character_event = { id = expd_purchase_claims.1000 }
					}
				}
			}
		}
		any_claimant = {
			limit = {
				expd_purchase_claims_this_character_is_a_notifiable_player = yes
				NOT = { top_liege = { any_realm_title = { title = PREVPREVPREV } } }
			}
			custom_tooltip = {
				text = expd_purchase_claims_this_player_is_notified_tt
				character_event = { id = expd_purchase_claims.1000 }
			}
		}
	}
}

# NB: To avoid multiplayer sync issues, changes in global flags with instant effect must be made by character_event on all players.

expd_purchase_claims_cost_rule_cycle = {
	if = {
		limit = { expd_purchase_claims_cost_is_standard_actual = yes }
		custom_tooltip = {
			text = expd_purchase_claims_cost_rule_change_to_half_tt
			expd_purchase_claims_cost_rule_clear = yes
			                                                      character_event = { id = expd_purchase_claims_global.9003 }
			any_player = { limit = { NOT = { character = PREV } } character_event = { id = expd_purchase_claims_global.9003 } }
		}
	}
	else_if = {
		limit = { expd_purchase_claims_cost_is_half = yes }
		custom_tooltip = {
			text = expd_purchase_claims_cost_rule_change_to_quarter_tt
			expd_purchase_claims_cost_rule_clear = yes
			                                                      character_event = { id = expd_purchase_claims_global.9004 }
			any_player = { limit = { NOT = { character = PREV } } character_event = { id = expd_purchase_claims_global.9004 } }
		}
	}
	else_if = {
		limit = { expd_purchase_claims_cost_is_quarter = yes }
		custom_tooltip = {
			text = expd_purchase_claims_cost_rule_change_to_tenth_tt
			expd_purchase_claims_cost_rule_clear = yes
			                                                      character_event = { id = expd_purchase_claims_global.9005 }
			any_player = { limit = { NOT = { character = PREV } } character_event = { id = expd_purchase_claims_global.9005 } }
		}
	}
	else_if = {
		limit = { expd_purchase_claims_cost_is_tenth = yes }
		custom_tooltip = {
			text = expd_purchase_claims_cost_rule_change_to_double_tt
			expd_purchase_claims_cost_rule_clear = yes
			                                                      character_event = { id = expd_purchase_claims_global.9001 }
			any_player = { limit = { NOT = { character = PREV } } character_event = { id = expd_purchase_claims_global.9001 } }
		}
	}
	else = { # Double or inconsistent
		custom_tooltip = {
			text = expd_purchase_claims_cost_rule_change_to_standard_tt
			expd_purchase_claims_cost_rule_clear = yes
			                                                      character_event = { id = expd_purchase_claims_global.9002 }
			any_player = { limit = { NOT = { character = PREV } } character_event = { id = expd_purchase_claims_global.9002 } }
		}
	}
}
